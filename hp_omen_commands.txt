HP OMEN Linux Module Commands
=============================

This file contains all the useful commands for controlling your HP OMEN system
after installing the custom hp-wmi kernel module.

INSTALLATION COMMANDS
====================

# Compile the module
cd /home/shaw/hp-omen-linux-module/src && make clean && make

# Install with DKMS (run as root)
sudo dkms install /home/shaw/hp-omen-linux-module

# Or manually load the module
sudo rmmod hp_wmi                    # Unload old module
sudo insmod hp-wmi.ko               # Load new module

# Check if module is loaded
lsmod | grep hp_wmi

RGB KEYBOARD CONTROL
===================

# Check current RGB settings for all zones
for zone in /sys/devices/platform/hp-wmi/rgb_zones/zone*; do echo "$(basename $zone):"; cat "$zone"; done

# Set all zones to blue
echo "0000FF" | sudo tee /sys/devices/platform/hp-wmi/rgb_zones/zone*

# Different shades of blue
echo "000080" | sudo tee /sys/devices/platform/hp-wmi/rgb_zones/zone*  # Dark blue
echo "0000FF" | sudo tee /sys/devices/platform/hp-wmi/rgb_zones/zone*  # Pure blue  
echo "4169E1" | sudo tee /sys/devices/platform/hp-wmi/rgb_zones/zone*  # Royal blue
echo "87CEEB" | sudo tee /sys/devices/platform/hp-wmi/rgb_zones/zone*  # Sky blue
echo "ADD8E6" | sudo tee /sys/devices/platform/hp-wmi/rgb_zones/zone*  # Light blue

# Other common colors
echo "FF0000" | sudo tee /sys/devices/platform/hp-wmi/rgb_zones/zone*  # Red
echo "00FF00" | sudo tee /sys/devices/platform/hp-wmi/rgb_zones/zone*  # Green
echo "FFFF00" | sudo tee /sys/devices/platform/hp-wmi/rgb_zones/zone*  # Yellow
echo "FF00FF" | sudo tee /sys/devices/platform/hp-wmi/rgb_zones/zone*  # Magenta
echo "00FFFF" | sudo tee /sys/devices/platform/hp-wmi/rgb_zones/zone*  # Cyan
echo "FFFFFF" | sudo tee /sys/devices/platform/hp-wmi/rgb_zones/zone*  # White
echo "FFA500" | sudo tee /sys/devices/platform/hp-wmi/rgb_zones/zone*  # Orange
echo "800080" | sudo tee /sys/devices/platform/hp-wmi/rgb_zones/zone*  # Purple
echo "FFC0CB" | sudo tee /sys/devices/platform/hp-wmi/rgb_zones/zone*  # Pink

# Set individual zones to different colors (rainbow effect)
echo "FF0000" | sudo tee /sys/devices/platform/hp-wmi/rgb_zones/zone00  # Zone 1: Red
echo "00FF00" | sudo tee /sys/devices/platform/hp-wmi/rgb_zones/zone01  # Zone 2: Green  
echo "0000FF" | sudo tee /sys/devices/platform/hp-wmi/rgb_zones/zone02  # Zone 3: Blue
echo "FFFF00" | sudo tee /sys/devices/platform/hp-wmi/rgb_zones/zone03  # Zone 4: Yellow

# Gaming theme (red/black)
echo "FF0000" | sudo tee /sys/devices/platform/hp-wmi/rgb_zones/zone00 zone02 zone03 zone01 # Zones 1&3: Red
echo "000000" | sudo tee /sys/devices/platform/hp-wmi/rgb_zones/zone01 zone03  # Zones 2&4: Off

# Individual zone control
echo "FF0000" | sudo tee /sys/devices/platform/hp-wmi/rgb_zones/zone00  # Zone 1 only
echo "00FF00" | sudo tee /sys/devices/platform/hp-wmi/rgb_zones/zone01  # Zone 2 only
echo "0000FF" | sudo tee /sys/devices/platform/hp-wmi/rgb_zones/zone02  # Zone 3 only
echo "FFFF00" | sudo tee /sys/devices/platform/hp-wmi/rgb_zones/zone03  # Zone 4 only

THERMAL PROFILE CONTROL
=======================

# Check available thermal profiles
cat /sys/devices/platform/hp-wmi/platform-profile/platform-profile-0/choices

# Get current thermal profile
cat /sys/devices/platform/hp-wmi/platform-profile/platform-profile-0/profile

# Set thermal profiles (requires root)
echo "performance" | sudo tee /sys/devices/platform/hp-wmi/platform-profile/platform-profile-0/profile
echo "balanced" | sudo tee /sys/devices/platform/hp-wmi/platform-profile/platform-profile-0/profile
echo "cool" | sudo tee /sys/devices/platform/hp-wmi/platform-profile/platform-profile-0/profile

FAN CONTROL
==========

# Check fan speeds (RPM)
cat /sys/devices/platform/hp-wmi/hwmon/hwmon*/fan1_input
cat /sys/devices/platform/hp-wmi/hwmon/hwmon*/fan2_input

# Check all fan speeds at once
ls /sys/devices/platform/hp-wmi/hwmon/hwmon*/fan*_input | xargs -I {} sh -c 'echo "$(basename {}): $(cat {})"'

# Check fan control mode
cat /sys/devices/platform/hp-wmi/hwmon/hwmon*/pwm1_enable

# Set fan control mode (requires root)
echo 2 | sudo tee /sys/devices/platform/hp-wmi/hwmon/hwmon*/pwm1_enable  # Automatic mode
echo 0 | sudo tee /sys/devices/platform/hp-wmi/hwmon/hwmon*/pwm1_enable  # Max speed mode

SYSTEM INFORMATION
=================

# Check hardware temperature
cat /sys/devices/platform/hp-wmi/hddtemp

# Check ambient light sensor
cat /sys/devices/platform/hp-wmi/als

# Set ambient light sensor (requires root)
echo 1 | sudo tee /sys/devices/platform/hp-wmi/als  # Enable
echo 0 | sudo tee /sys/devices/platform/hp-wmi/als  # Disable

# Check dock status
cat /sys/devices/platform/hp-wmi/dock

# Check tablet mode
cat /sys/devices/platform/hp-wmi/tablet

# Check display status
cat /sys/devices/platform/hp-wmi/display

# Check POST error code
cat /sys/devices/platform/hp-wmi/postcode

# Clear POST error code (requires root)
echo true | sudo tee /sys/devices/platform/hp-wmi/postcode

MONITORING SCRIPTS
=================

# Monitor all system status
watch -n 1 'echo "=== Thermal Profile ===" && cat /sys/devices/platform/hp-wmi/platform-profile/platform-profile-0/profile && echo -e "\n=== Fan Speeds ===" && ls /sys/devices/platform/hp-wmi/hwmon/hwmon*/fan*_input | xargs -I {} sh -c "echo \"\$(basename {}): \$(cat {})\"" && echo -e "\n=== Temperature ===" && cat /sys/devices/platform/hp-wmi/hddtemp'

# Monitor RGB settings
watch -n 2 'for zone in /sys/devices/platform/hp-wmi/rgb_zones/zone*; do echo "$(basename $zone):"; cat "$zone"; echo; done'

# Simple fan monitor
watch -n 1 'echo "Fan1: $(cat /sys/devices/platform/hp-wmi/hwmon/hwmon*/fan1_input) RPM" && echo "Fan2: $(cat /sys/devices/platform/hp-wmi/hwmon/hwmon*/fan2_input) RPM"'

AUTOMATIC RGB SETUP ON STARTUP
===============================

# Set up automatic RGB color restoration on login
./install_rgb_autostart.sh

# Manually run the RGB setup script
./setup_rgb_colors.sh

# Set all zones to blue on startup
./setup_rgb_colors.sh --color 0000FF

# Set rainbow colors with verification
./setup_rgb_colors.sh --verify

# Check status of the auto-setup service
systemctl --user status hp-omen-rgb.service

# View logs from the auto-setup service
journalctl --user -u hp-omen-rgb.service

# Manually start the service
systemctl --user start hp-omen-rgb.service

# Disable automatic startup
systemctl --user disable hp-omen-rgb.service
